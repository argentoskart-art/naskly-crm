function searchCustomerById(id) {
  const sheet = getMainSheet_();
  const row = findRowById_(sheet, normalizeId_(id));
  if (!row) return { found: false };
  return { found: true, record: rowToRecord_(sheet, row), row };
}

function getCustomersByStatus(status) {
  const sheet = getMainSheet_();
  const lastRow = sheet.getLastRow();
  const out = [];
  for (let r = BACKEND_START_ROW; r <= lastRow; r++) {
    const current = String(sheet.getRange(r, 14).getValue() || '').trim();
    if (current === status) out.push(rowToRecord_(sheet, r));
  }
  return out;
}

function saveCustomerRecord(payload) {
  const sheet = getMainSheet_();
  const record = normalizePayload_(payload || {});
  const existingRow = Number.isInteger(payload && payload.row) ? payload.row : (record.id ? findRowById_(sheet, record.id) : null);
  const row = existingRow || findNextEmptyRow_(sheet);
  const isNew = !existingRow;

  if (isNew && !record.id) record.id = generateNextId_(sheet);

  if (record.paymentStatus === 'تم بالكامل') {
    const total = Number(record.totalPrice);
    const paid = Number(record.paid);
    const remaining = Number(record.remaining);
    if (Number.isFinite(remaining) && remaining !== 0) {
      record.paid = (Number.isFinite(paid) ? paid : 0) + remaining;
      record.remaining = 0;
    } else if (Number.isFinite(total) && Number.isFinite(paid) && paid < total) {
      record.paid = total;
      record.remaining = 0;
    }
  }

  const rowValues = [
    record.id || '',
    record.bookingDate || '',
    record.deliveryDate || '',
    record.clientName || '',
    record.whatsapp || '',
    record.source || '',
    record.dealOwner || '',
    record.deliveryStaff || '',
    record.serviceType || '',
    record.serviceSpecs || '',
    record.totalPrice === '' ? '' : Number(record.totalPrice),
    record.paid === '' ? '' : Number(record.paid),
    record.remaining === '' ? '' : Number(record.remaining),
    record.deliveryStatus || '',
    record.paymentStatus || '',
  ];

  sheet.getRange(row, 1, 1, 15).setValues([rowValues]);
  sheet.getRange(row, 2, 1, 2).setNumberFormat('yyyy-mm-dd');
  sheet.getRange(row, 11, 1, 3).setNumberFormat('#,##0.00');
  sheet.getRange(row, 13).setFormula(`=IF(OR(K${row}="",L${row}=""),"",MAX(K${row}-L${row},0))`);

  return { ok: true, row, id: record.id, isNew };
}

function rowToRecord_(sheet, row) {
  const range = sheet.getRange(row, 1, 1, 15);
  const values = range.getValues()[0];
  const displayValues = range.getDisplayValues()[0];

  return {
    row,
    id: displayValues[0] ? displayValues[0].toString().trim() : (values[0] || ''),
    bookingDate: toIsoDate_(values[1]),
    deliveryDate: toIsoDate_(values[2]),
    clientName: displayValues[3] ? displayValues[3].toString().trim() : (values[3] || ''),
    whatsapp: displayValues[4] ? displayValues[4].toString().trim() : (values[4] ? String(values[4]).trim() : ''),
    source: displayValues[5] ? displayValues[5].toString().trim() : (values[5] || ''),
    dealOwner: displayValues[6] ? displayValues[6].toString().trim() : (values[6] || ''),
    deliveryStaff: displayValues[7] ? displayValues[7].toString().trim() : (values[7] || ''),
    serviceType: displayValues[8] ? displayValues[8].toString().trim() : (values[8] || ''),
    serviceSpecs: displayValues[9] ? displayValues[9].toString().trim() : (values[9] || ''),
    totalPrice: values[10] === '' ? '' : Number(values[10]),
    paid: values[11] === '' ? '' : Number(values[11]),
    remaining: values[12] === '' ? '' : Number(values[12]),
    deliveryStatus: displayValues[13] ? displayValues[13].toString().trim() : (values[13] || ''),
    paymentStatus: displayValues[14] ? displayValues[14].toString().trim() : (values[14] || ''),
  };
}

function normalizePayload_(payload) {
  const serviceType = Array.isArray(payload.serviceType)
    ? payload.serviceType.map(cleanText_).filter(Boolean).join(', ')
    : cleanText_(payload.serviceType);
  return {
    id: normalizeId_(payload.id),
    bookingDate: normalizeDate_(payload.bookingDate),
    deliveryDate: normalizeDate_(payload.deliveryDate),
    clientName: cleanText_(payload.clientName),
    whatsapp: cleanText_(payload.whatsapp),
    source: cleanText_(payload.source),
    dealOwner: cleanText_(payload.dealOwner),
    deliveryStaff: cleanText_(payload.deliveryStaff),
    serviceType,
    serviceSpecs: cleanText_(payload.serviceSpecs),
    totalPrice: normalizeNumber_(payload.totalPrice),
    paid: normalizeNumber_(payload.paid),
    remaining: normalizeNumber_(payload.remaining),
    deliveryStatus: cleanText_(payload.deliveryStatus),
    paymentStatus: cleanText_(payload.paymentStatus),
  };
}

function findRowById_(sheet, id) {
  if (!id) return null;
  const lastRow = sheet.getLastRow();
  if (lastRow < BACKEND_START_ROW) return null;
  const values = sheet.getRange(BACKEND_START_ROW, 1, lastRow - BACKEND_START_ROW + 1, 1).getValues().flat();
  for (let i = 0; i < values.length; i++) {
    if (normalizeId_(values[i]) === id) return BACKEND_START_ROW + i;
  }
  return null;
}

function findNextEmptyRow_(sheet) {
  return Math.max(sheet.getLastRow() + 1, BACKEND_START_ROW);
}

function generateNextId_(sheet) {
  const lastRow = sheet.getLastRow();
  const end = Math.max(lastRow, BACKEND_START_ROW);
  const values = sheet.getRange(BACKEND_START_ROW, 1, end - BACKEND_START_ROW + 1, 1).getValues().flat();
  let maxNum = 0;
  values.forEach(v => {
    const m = String(v || '').trim().match(/^N(\d+)$/i);
    if (m) maxNum = Math.max(maxNum, parseInt(m[1], 10));
  });
  return `${BACKEND_ID_PREFIX}${String(maxNum + 1).padStart(3, '0')}`;
}

function getTeamNames_() {
  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName(BACKEND_TEAM_SHEET_NAME);
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  return sheet.getRange(2, 1, lastRow - 1, 1).getValues().flat().filter(Boolean).map(String);
}

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];

}

function getLogoDataUrl_() {
  try {
    const fileId = '1FCotweW1r30F-275wwteCdoqsWX3XWNY';
    const blob = DriveApp.getFileById(fileId).getBlob();
    const bytes = blob.getBytes();

function getSpreadsheet_() {

function getMainSheet_() {
  const ss = getSpreadsheet_();
  const sheet = ss.getSheetByName(BACKEND_MAIN_SHEET_NAME);
  if (!sheet) throw new Error('MainData sheet not found');
  return sheet;
}

function normalizeId_(value) {
  return String(value || '').trim().toUpperCase();
}

function cleanText_(value) {
  return String(value == null ? '' : value).trim();
}

function normalizeNumber_(value) {
  if (value === '' || value == null) return '';
  const n = Number(value);
  return Number.isFinite(n) ? n : '';
}

function normalizeDate_(value) {
  if (!value) return '';
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value)) return value;
  const d = new Date(value);
  return isNaN(d) ? '' : d;
}

function toIsoDate_(value) {
  if (!value) return '';
  const d = new Date(value);
  if (isNaN(d)) return '';
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

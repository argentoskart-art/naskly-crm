  <script>
    const state = {
      fields: [],
      options: { statuses: [], sources: [], teams: [], services: [] },
      record: {}
    };

    const baseInput = 'w-full bg-palette-navy-dark border border-palette-navy-light focus:border-palette-gold focus:ring-1 focus:ring-palette-gold text-palette-ice text-sm rounded-xl px-3.5 py-2.5 outline-none transition';
    const labelCls = 'mb-1.5 block text-xs font-semibold text-palette-ice-muted whitespace-nowrap';
    const fieldWrap = 'space-y-1';

    function setMessage(text) { 
      const msgEl = document.getElementById('message');
      if (msgEl) msgEl.textContent = text || ''; 
    }

    function setLoading(on) {
      document.body.classList.toggle('opacity-75', !!on);
      document.body.classList.toggle('pointer-events-none', !!on);
    }

    function escapeHtml(value) {
      return String(value ?? '')
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function getFieldOptions(field) {
      if (field.options) return field.options;
      if (field.optionsSource === 'team') return state.options.teams || [];
      if (field.optionsSource === 'services') return state.options.services || [];
      return [];
    }

    function getMultiValues(key) {
      const raw = state.record[key];
      if (Array.isArray(raw)) return raw.filter(Boolean);
      return String(raw || '').split(',').map(v => v.trim()).filter(Boolean);
    }

    function setMultiValues(key, values) {
      state.record[key] = values.filter(Boolean);
      const text = document.getElementById(key + '__text');
      if (text) text.value = values.join(', ');
      const hidden = document.getElementById(key + '__hidden');
      if (hidden) hidden.value = values.join(', ');
      const chips = document.getElementById(key + '__chips');
      if (chips) chips.innerHTML = renderChipsHtml(key, values);
      const query = document.getElementById(key + '__query');
      if (query) query.value = '';
      const panel = document.getElementById(key + '__panel');
      if (panel) panel.classList.add('hidden');
    }

    function renderChipsHtml(key, values) {
      if (!values.length) return '<span class="text-xs text-palette-ice-muted">لم يتم اختيار أي خدمة بعد</span>';
      return values.map(v => `
        <span class="inline-flex items-center gap-2 rounded-lg bg-palette-green/20 border border-palette-green/40 px-3 py-1 text-xs font-semibold text-palette-green whitespace-nowrap">
          <span>${escapeHtml(v)}</span>
          <button type="button" class="multi-remove text-palette-orange hover:text-palette-gold text-base leading-none" data-key="${key}" data-value="${escapeHtml(v)}">×</button>
        </span>
      `).join('');
    }

    function removeMultiValue(key, value) {
      const values = getMultiValues(key).filter(v => v !== value);
      setMultiValues(key, values);
      state.record[key] = values;
    }

    function filterServices(query) {
      const q = String(query || '').trim().toLowerCase();
      return getFieldOptions({ optionsSource: 'services' }).filter(item => item.toLowerCase().includes(q));
    }

    function openMultiPanel(key) {
      const panel = document.getElementById(key + '__panel');
      if (panel) panel.classList.remove('hidden');
      const query = document.getElementById(key + '__query');
      if (query) query.focus();
    }

    function renderMultiSelectField(field) {
      const values = getMultiValues(field.key);
      const chips = renderChipsHtml(field.key, values);
      const label = `<label class="${labelCls}">${field.label}</label>`;
      return `
        <div class="${fieldWrap} md:col-span-2 relative">
          ${label}
          <div class="space-y-3 rounded-2xl border border-palette-navy-light/60 bg-palette-navy-dark/90 p-3.5">
            <div class="space-y-2">
              <div class="text-xs font-medium text-palette-ice-muted">الخدمات المختارة:</div>
              <div class="flex flex-wrap gap-2" id="${field.key}__chips">${chips}</div>
            </div>
            <input id="${field.key}__text" type="hidden" value="${escapeHtml(values.join(', '))}">
            <input id="${field.key}__hidden" type="hidden" value="${escapeHtml(values.join(', '))}">
            <input id="${field.key}__query" class="${baseInput}" type="text" placeholder="اكتب للبحث في قائمة الخدمات..." oninput="updateMultiSuggestions('${field.key}', this.value)" onclick="openMultiPanel('${field.key}')" autocomplete="off">
            <div id="${field.key}__panel" class="hidden rounded-xl border border-palette-navy-light/60 bg-palette-navy p-2 shadow-xl">
              <div class="mb-2 text-xs font-semibold text-palette-gold">قائمة الخدمات المتاحة:</div>
              <div id="${field.key}__list" class="space-y-1 max-h-48 overflow-auto"></div>
            </div>
          </div>
        </div>
      `;
    }

    function updateMultiSuggestions(key, query) {
      const list = document.getElementById(key + '__list');
      const panel = document.getElementById(key + '__panel');
      const selected = getMultiValues(key);
      const matches = filterServices(query).filter(item => !selected.includes(item));
      if (!panel) return;
      panel.classList.remove('hidden');
      if (!matches.length) {
        list.innerHTML = '<div class="px-3 py-2 text-xs text-palette-ice-muted">لا توجد نتائج متطابقة.</div>';
        return;
      }
      list.innerHTML = matches.map(item => `
        <button type="button" class="service-option w-full rounded-lg px-3 py-2 text-right text-xs font-semibold text-palette-ice hover:bg-palette-navy-light transition whitespace-nowrap" data-key="${key}" data-value="${escapeHtml(item)}">
          ${escapeHtml(item)}
        </button>
      `).join('');
      attachServiceOptionHandlers();
    }

    function pickMultiValue(key, value) {
      const values = getMultiValues(key);
      if (!values.includes(value)) values.push(value);
      setMultiValues(key, values);
      state.record[key] = values;
      const query = document.getElementById(key + '__query');
      if (query) query.focus();
      updateMultiSuggestions(key, query ? query.value : '');
    }

    function attachServiceOptionHandlers() {
      document.querySelectorAll('.service-option').forEach(btn => {
        if (btn.dataset.bound === '1') return;
        btn.dataset.bound = '1';
        btn.addEventListener('mousedown', ev => ev.preventDefault());
        btn.addEventListener('click', () => {
          const key = btn.dataset.key;
          const value = btn.dataset.value;
          if (key && value) pickMultiValue(key, value);
        });
      });
      document.querySelectorAll('.multi-remove').forEach(btn => {
        if (btn.dataset.bound === '1') return;
        btn.dataset.bound = '1';
        btn.addEventListener('mousedown', ev => ev.preventDefault());
        btn.addEventListener('click', () => {
          const key = btn.dataset.key;
          const value = btn.dataset.value;
          if (key && value) removeMultiValue(key, value);
        });
      });
    }

    function renderField(field) {
      const value = state.record[field.key] ?? '';
      const label = `<label class="${labelCls}" for="${field.key}">${field.label}</label>`;
      if (field.readonly) return `<div class="${fieldWrap}">${label}<input id="${field.key}" class="${baseInput} opacity-70 cursor-not-allowed" type="text" value="${escapeHtml(value)}" readonly></div>`;
      if (field.type === 'date') return `<div class="${fieldWrap}">${label}<input id="${field.key}" class="${baseInput}" type="date" value="${escapeHtml(value)}"></div>`;
      if (field.type === 'number') return `<div class="${fieldWrap}">${label}<input id="${field.key}" class="${baseInput}" type="number" step="0.01" value="${escapeHtml(value)}"></div>`;
      if (field.type === 'multi-select') return renderMultiSelectField(field);
      if (field.type === 'select') {
        const opts = getFieldOptions(field).map(opt => `<option value="${escapeHtml(opt)}" ${String(opt) === String(value) ? 'selected' : ''}>${escapeHtml(opt)}</option>`).join('');
        return `<div class="${fieldWrap}">${label}<select id="${field.key}" class="${baseInput}"><option value="">-- اختر --</option>${opts}</select></div>`;
      }
      if (field.key === 'serviceSpecs') return `<div class="space-y-1 md:col-span-2">${label}<textarea id="${field.key}" class="${baseInput} min-h-24 resize-y">${escapeHtml(value)}</textarea></div>`;
      return `<div class="${fieldWrap}">${label}<input id="${field.key}" class="${baseInput}" type="text" value="${escapeHtml(value)}"></div>`;
    }

    function getSectionForField(field) {
      if (['id', 'clientName', 'whatsapp', 'source'].includes(field.key)) return 'customerFields';
      if (['bookingDate', 'deliveryDate', 'dealOwner', 'deliveryStaff', 'serviceType', 'serviceSpecs'].includes(field.key)) return 'workFields';
      if (['totalPrice', 'paid', 'remaining', 'deliveryStatus', 'paymentStatus'].includes(field.key)) return 'paymentFields';
      return 'workFields';
    }

    function renderForm() {
      const buckets = { customerFields: [], workFields: [], paymentFields: [] };
      state.fields.forEach(field => {
        buckets[getSectionForField(field)].push(renderField(field));
      });
      document.getElementById('customerFields').innerHTML = buckets.customerFields.join('');
      document.getElementById('workFields').innerHTML = buckets.workFields.join('');
      document.getElementById('paymentFields').innerHTML = buckets.paymentFields.join('');
      state.fields.forEach(field => {
        const el = document.getElementById(field.key);
        if (!el || field.readonly) return;
        if (field.type === 'multi-select') return;
        el.addEventListener('input', () => state.record[field.key] = el.value);
        el.addEventListener('change', () => state.record[field.key] = el.value);
      });
      attachServiceOptionHandlers();
    }

    function fillForm(record) {
      state.record = Object.assign({
        id: '', bookingDate: '', deliveryDate: '', clientName: '', whatsapp: '', source: '',
        dealOwner: '', deliveryStaff: '', serviceType: [], serviceSpecs: '', totalPrice: '',
        paid: '', remaining: '', deliveryStatus: '', paymentStatus: '', row: ''
      }, record || {});
      renderForm();
      state.fields.forEach(field => {
        const el = document.getElementById(field.key);
        if (!el) return;
        if (field.type === 'multi-select') {
          setMultiValues(field.key, getMultiValues(field.key));
        } else {
          el.value = state.record[field.key] ?? '';
        }
      });
      document.getElementById('searchId').value = state.record.id || '';
    }

    function syncFormToState() {
      state.fields.forEach(field => {
        const el = document.getElementById(field.key);
        if (!el) return;
        if (field.type === 'multi-select') {
          state.record[field.key] = getMultiValues(field.key);
        } else {
          state.record[field.key] = el.value;
        }
      });
    }

    function clearForm() {
      fillForm({});
      setMessage('تم تجهيز النموذج لإضافة عميل جديد.');
      document.getElementById('results').innerHTML = '<div class="p-6 text-center text-sm text-palette-ice-muted">جاهز لإضافة عميل جديد.</div>';
      document.getElementById('resultsHint').textContent = 'لم يتم البحث بعد.';
    }

    function searchById() {
      const id = document.getElementById('searchId').value.trim();
      if (!id) { setMessage('يرجى إدخال الرقم التعريفي أولاً.'); return; }
      setLoading(true);
      google.script.run
        .withSuccessHandler(res => {
          setLoading(false);
          if (!res || !res.found) {
            setMessage('لم يتم العثور على عميل بهذا الرقم التعريفي.');
            document.getElementById('results').innerHTML = '<div class="p-6 text-center text-sm text-palette-ice-muted">لا توجد نتائج تطابق هذا الرقم.</div>';
            document.getElementById('resultsHint').textContent = 'لا توجد نتائج.';
            return;
          }
          fillForm(res.record);
          setMessage(`تم تحميل بيانات العميل ${res.record.id} بنجاح.`);
          renderSingleResult(res.record);
          document.getElementById('resultsHint').textContent = 'نتيجة البحث الحالية.';
        })
        .withFailureHandler(err => {
          setLoading(false);
          setMessage(err.message || String(err));
        })
        .searchCustomerById(id);
    }

    function showStatus(status) {
      setLoading(true);
      google.script.run
        .withSuccessHandler(rows => {
          setLoading(false);
          renderStatusResults(status, rows || []);
          setMessage(`عرض العملاء بالحالة: ${status}`);
          document.getElementById('resultsHint').textContent = `تصفية بحسب الحالة: ${status}`;
        })
        .withFailureHandler(err => {
          setLoading(false);
          setMessage(err.message || String(err));
        })
        .getCustomersByStatus(status);
    }

    function saveRecord() {
      syncFormToState();
      setLoading(true);
      google.script.run
        .withSuccessHandler(res => {
          setLoading(false);
          if (!res || !res.ok) {
            setMessage('فشل حفظ البيانات.');
            return;
          }
          state.record.id = res.id;
          state.record.row = res.row || state.record.row;
          fillForm(state.record);
          setMessage(res.isNew ? `تم حفظ العميل الجديد ${res.id} بنجاح.` : `تم تحديث بيانات العميل ${res.id} بنجاح.`);
        })
        .withFailureHandler(err => {
          setLoading(false);
          setMessage(err.message || String(err));
        })
        .saveCustomerRecord(state.record);
    }

    function renderSingleResult(record) { renderResultsTable([record]); }
    function renderStatusResults(status, rows) { renderResultsTable(rows); }

    function renderResultsTable(rows) {
      if (!rows || !rows.length) {
        document.getElementById('results').innerHTML = '<div class="p-6 text-center text-sm text-palette-ice-muted">لا توجد بيانات مطابقة.</div>';
        return;
      }
      const headers = ['ID', 'تاريخ الحجز', 'تاريخ الاستلام', 'اسم العميل', 'الواتساب', 'المصدر', 'مسؤول الديل', 'موظف التسليم', 'الخدمات', 'المواصفات', 'إجمالي السعر', 'المدفوع', 'المتبقي', 'حالة التسليم', 'حالة المدفوعات'];
      const body = rows.map(r => {
        const waText = r.whatsapp ? String(r.whatsapp).trim() : '';
        const waClean = waText.replace(/\D/g, '');
        const waButton = waText ? 
          `<a href="https://wa.me/${waClean.startsWith('2') ? waClean : (waClean.length === 10 || waClean.length === 11 ?

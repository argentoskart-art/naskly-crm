const REMINDER_SPREADSHEET_ID = '1NQgOnxdCXSlPjDU-hmYUOA2kINd1MgnX3OsgwFpfH8A';
const REMINDER_MAIN_SHEET_NAME = 'MainData';
const REMINDER_START_ROW = 3;
const REMINDER_LOGO_FILE_ID = '1FCotweW1r30F-275wwteCdoqsWX3XWNY';
const REMINDER_RECIPIENT = 'nsqlycorp@gmail.com';

function sendTomorrowDeliveryReminderEmail() {
  const rows = getTomorrowDeliveryReminderRows_();
  const subject = 'تنويه: عملاء موعد تسليمهم غدا';
  const htmlBody = buildReminderEmailHtml_(rows);
  const plainBody = buildReminderPlainText_(rows);

  MailApp.sendEmail({
    to: REMINDER_RECIPIENT,
    subject: subject,
    body: plainBody,
    htmlBody: htmlBody,
  });

  return { ok: true, count: rows.length, recipient: REMINDER_RECIPIENT, subject: subject };
}

function getTomorrowDeliveryReminderRows_() {
  const sheet = SpreadsheetApp.openById(REMINDER_SPREADSHEET_ID).getSheetByName(REMINDER_MAIN_SHEET_NAME);
  if (!sheet) throw new Error('MainData sheet not found');

  const lastRow = sheet.getLastRow();
  if (lastRow < REMINDER_START_ROW) return [];

  const values = sheet.getRange(REMINDER_START_ROW, 1, lastRow - REMINDER_START_ROW + 1, 15).getValues();
  const tomorrowKey = Utilities.formatDate(addDays_(new Date(), 1), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const openStatuses = new Set(['لم تبدأ بعد', 'جزئي']);
  const rows = [];

  values.forEach((row, index) => {
    const deliveryDate = toIsoDate_(row[2]);
    const deliveryStatus = String(row[13] || '').trim();
    if (deliveryDate && deliveryDate <= tomorrowKey && openStatuses.has(deliveryStatus)) {
      rows.push({
        rowNumber: REMINDER_START_ROW + index,
        id: row[0] || '',
        clientName: row[3] || '',
        deliveryDate: deliveryDate,
        deliveryStatus: deliveryStatus,
        whatsapp: row[4] || '',
        totalPrice: row[10] || '',
        paid: row[11] || '',
        remaining: row[12] || '',
        paymentStatus: row[14] || '',
      });
    }
  });

  return rows;
}

function buildReminderPlainText_(rows) {
  const displayRows = rows.slice(0, 20);
  const lines = [
    'تنويه: هذه قائمة بالعملاء ميعاد تسليمهم غدا وحالتهم مازالت مفتوحه يرجي اتخاذ اللازم.',
    ''
  ];

  if (!displayRows.length) {
    lines.push('لا يوجد عملاء يستحقون التذكير غدا.');
    return lines.join('\n');
  }

  displayRows.forEach((item, index) => {
    lines.push(
      `${index + 1}. ID: ${item.id} | الاسم: ${item.clientName} | الاستلام: ${item.deliveryDate} | الحالة: ${item.deliveryStatus} | الواتساب: ${item.whatsapp} | الإجمالي: ${item.totalPrice} | المدفوع: ${item.paid} | المتبقي: ${item.remaining} | حالة المدفوعات: ${item.paymentStatus}`
    );
  });

  if (rows.length > displayRows.length) {
    lines.push('');
    lines.push(`تم عرض أول ${displayRows.length} نتيجة فقط من أصل ${rows.length}.`);
  }

  return lines.join('\n');
}

function getReminderLogoDataUrl_() {
  const blob = DriveApp.getFileById(REMINDER_LOGO_FILE_ID).getBlob();
  return 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes());
}

function addDays_(date, days) {
  const d = new Date(date);
  d.setDate(d.getDate() + days);
  return d;
}

function toIsoDate_(value) {
  if (!value) return '';
  const d = new Date(value);
  if (isNaN(d)) return '';
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

function escapeHtml_(value) {
  return String(value == null ? '' : value)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}
